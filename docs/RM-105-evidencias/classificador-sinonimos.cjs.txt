function lev(a, b) { const m = a.length, n = b.length; const d = Array.from({ length: m + 1 }, (_, i) => [i, ...Array(n).fill(0)]); for (let j = 0; j <= n; j++) d[0][j] = j; for (let i = 1; i <= m; i++) for (let j = 1; j <= n; j++) d[i][j] = Math.min(d[i - 1][j] + 1, d[i][j - 1] + 1, d[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1)); return d[m][n]; }
const has = (t, list) => { const s = W(t); return list.some(w => s.includes(W(w))); };
const eq = (t, list) => list.includes(t);

const byId = new Map(E.map(e => [e.id, e]));
function classificar(r, ignorarPre) {
  if (r.cat && !ignorarPre) return r;
  const t = r.norm, e = byId.get(r.ent), nome = strip(e.name).replace(/[^a-z0-9 ]/g, ' ').replace(/\s+/g, ' ').trim();
  const nomeW = nome.split(' ')[0];
  const mk = (cat, conf, ev) => ({ ...r, cat, conf, ev });
  // a) abreviação inequívoca
  if (ABREV[t] !== undefined && ABREV[t]) return mk('abreviacao_inequivoca', 'alta', `sigla de ${ABREV[t]}`);
  // b) DCI estrangeiro documentado
  if (DCI_ALT[t] && (nome.includes(DCI_ALT[t].split(' ')[0]) || strip(e.fullName || '').includes(DCI_ALT[t].split(' ')[0]))) return mk('sinonimo_farmacologico_valido', 'media', `DCI/USAN alternativo de ${e.name}`);
  // c) variante ortográfica de uma única palavra do próprio princípio ativo
  const palavrasNome = nome.split(' ').filter(w => w.length >= 5); const alvo = !t.includes(' ') && t.length >= 5 && !has(t, CLASSE) ? palavrasNome.find(w => t.slice(0, 4) === w.slice(0, 4) && lev(t, w) <= 3) : null;
  if (alvo) return mk('sinonimo_farmacologico_valido', 'media', `variante ortográfica/DCI de ${e.name} (lev=${lev(t, alvo)})`);
  // d) classe terapêutica (palavra de classe como termo inteiro ou como núcleo)
  if (eq(t, GENERICO)) return mk('termo_generico', 'media', 'termo genérico/contextual sem identidade farmacológica');
  if (eq(t, SINTOMA)) return mk('sintoma', 'alta', 'sintoma');
  if (eq(t, INDIC) || has(t, INDIC.filter(x => x.length >= 6))) {
    // indicação (se também citar uma classe, prevalece a classe abaixo)
    if (!has(t, CLASSE)) return mk('indicacao_clinica', 'media', 'indicação clínica / contexto de uso');
  }
  if (has(t, CLASSE)) return mk('classe_terapeutica', 'media', 'contém termo de classe terapêutica: ' + CLASSE.filter(w => W(t).includes(W(w))).slice(0, 2).join(', '));
  if (has(t, INDIC)) return mk('indicacao_clinica', 'baixa', 'contém termo de indicação clínica');
  return mk('nao_validado', 'baixa', 'sem evidência de equivalência verificável (possível nome comercial, sigla ou contexto)');
}
